གཉིས་པ་འདུས་བྱས་ཀྱི་ཆོས་ཀྱི་སྐྱེ་ཚུལ་ལ་གཉིས། དངོས་དང༌། ཞར་ལ་རྒྱུ་རྐྱེན་འབྲས་བུ་དང་བཅས་པ་བཤད་པའོ། །དང་པོ་(དངོས་)ལ་གཉིས། གཟུགས་ཅན་གྱི་ཆོས་ཀྱི་སྐྱེ་ཚུལ་དང༌། གཟུགས་ཅན་མ་ཡིན་པའི་ཆོས་ཀྱི་སྐྱེ་ཚུལ་ལོ། །དང་པོ་(གཟུགས་ཅན་གྱི་ཆོས་ཀྱི་སྐྱེ་ཚུལ་)ནི། འདུས་བྱས་རྣམས་མཚན་ཉིད་ཐ་དད་པ་བཞིན་དུ་ཐ་དད་པ་ཁོ་ནར་སྐྱེའམ། ངེས་པར་ལྷན་ཅིག་ཏུ་སྐྱེ་བ་ཡོད་ཅེ་ན། ལྷན་ཅིག་ཏུ་སྐྱེ་བ་ཡོད་དེ། སྤྱིར་ཆོས་རྣམས་གཞི་ལྔས་བསྡུས་ཏེ། གཟུགས་དང༌། སེམས་དང༌། སེམས་བྱུང་དང༌། ལྡན་མིན་འདུ་བྱེད་དང༌། འདུས་མ་བྱས་ཀྱི་གཞིས་བསྡུས་པའི་ཕྱིར། འདུས་མ་བྱས་ལ་སྐྱེ་བ་མེད་པས་ཆོས་གང་དང་ཡང་ལྷན་ཅིག་ཏུ་སྐྱེ་བ་མེད་ལ། གཞན་རྣམས་ཀྱི་སྐྱེ་ཚུལ་ལ། གཟུགས་ཅན་གྱི་སྐྱེ་ཚུལ་དང༌། གཟུགས་ཅན་མ་ཡིན་པའི་སྐྱེ་ཚུལ་གཉིས་སུ་ངེས་པས། དང་པོ་ལ། འདོ༷ད་ཁམས་ན༷་དབ༷ང་པོ༷འི་རྡུལ་མེད༷་པ༷་དང༷་སྒྲའི༷་རྡུལ་མེད༷་པའི་རྡུལ་ཕྲ༷་རབ༷་བསགས་པའི་གོང་བུ་ཆུང་ཤོས་ཉི་ཟེར་གྱི་རྡུལ་ལྟ་བུ་གཅིག་ཀྱང་རྡུལ༷་རྫས༷་བརྒྱད༷་དང་ལྷན་ཅིག་སྐྱེ་སྟེ། འབྱུང་བ་བཞི་དང་གཟུགས་དྲི་རོ་རེག་རྣམས་དང་ལྷན་ཅིག་སྐྱེ་བའི་ཕྱིར། ལུས༷་དབ༷ང་དང་ལྡན༷་པ་ལ༷་དེ་དང་རྫས༷་དགུ༷་ལྷན་ཅིག་སྐྱེ། མིག་སོགས་དབང་པོ་གཞན་གང་རུང་གཅིག་དང་ལྡན་པ་ལ་དགུ་པོ་དེ་དང་རྫས་བཅུ་ལྷན་ཅིག་སྐྱེ། སྒྲ་དང་ལྡན་ན་དེ་དང་རྫས་བཅུ་གཅིག་ལྷན་ཅིག་སྐྱེ། གཟུགས་ཁམས་ན་དྲི་རོ་གཉིས་མེད་པས། གཟུགས་ན་དབང་པོ་མེད་པ་དང༌། །སྒྲ་མེད་ཕྲ་རབ་རྡུལ་རྫས་དྲུག །ལུས་དབང་ལྡན་ལ་རྫས་བདུན་ནོ། །དབང་པོ་གཞན་ལྡན་རྫས་བརྒྱད་དོ། །ཞེས་བཤད་པར་བྱའོ། ། གཉིས་པ་གཟུགས་ཅན་མ་ཡིན་པའི་སྐྱེ་ཚུལ་ལ་གཉིས། བསྟན། བཤད་དོ། །དང་པོ་(མདོར་བསྟན་པ་)ནི། གཟུགས་ཅན་གྱི་སྐྱེ་ཚུལ་དེ་ལྟར་ཡིན་ན། གཟུགས་ཅན་མ་ཡིན་པའི་སྐྱེ་ཚུལ་ཇི་ལྟར་ཡིན་ཞེ་ན། སེམ༷ས་དང༷་སེམ༷ས་བྱུང༷་ངེ༷ས་པར་ལྷན༷་ཅིག༷་སྐྱེ་སྟེ། དེ་གཉིས་ཕན་ཚུན་ལྷན་ཅིག་འབྱུང་བའི་རྒྱུ་ཡིན་པའི་ཕྱིར། དེ་བཞིན་དུ་ཆོས་གང་སྐྱེ་བ་ཐམ༷ས་ཅད༷་ཀྱང་རང་འདུ༷ས་བྱས༷་སུ་མཚོན་པའི་མཚ༷ན་ཉིད༷་སྐྱེ་གནས་རྒ་འཇིག་བཞི་དང༷་ལྷན་ཅིག་འབྱུང་བ་དང༌། ཐོབ་བྱའི་ཆོས་རྒྱུད་དུ་གཏོགས་པ་དག་ནི་རང་དང་དུས་མཉམ་པའི་ཐོབ༷་པ༷་དག་དང་ལྷན་ཅིག་ཏུ་སྐྱེའོ། །འམ༷་ཞེས་པས་ཐོབ་བྱའི་ཆོས་རྒྱུད་དུ་མ་གཏོགས་པ་རྣམས་ནི་ཐོབ་པ་དང་ལྷན་ཅིག་སྐྱེ་བ་མ་ཡིན་པར་བསྟན་ཏེ། ཐོབ་བྱའི་ཆོས་རྒྱུད་དུ་མ་གཏོགས་པ་ཕལ་ཆེ་བ་ལ་ཐོབ་པ་མེད་པས་སོ། །འགོག་པ་གཉིས་ལ་ནི་ཐོབ་པ་ཡོད་ཀྱང་དེ་གཉིས་ལ་སྐྱེ་བ་མེད་པས་མ་སྨོས་སོ། ། གཉིས་པ་རྒྱས་བཤད་ལ་གཉིས། མཚུངས་ལྡན་གྱི་སྐྱེ་ཚུལ། མཚུངས་ལྡན་མ་ཡིན་པའི་སྐྱེ་ཚུལ་ལོ། །དང་པོ་(མཚུངས་ལྡན་གྱི་སྐྱེ་ཚུལ་)ལ་བཞི། སྡེ་ཚན་དུ་ངེས་པའི་རབ་དབྱེ། མ་ངེས་པ་དང་བཅས་པའི་མཚུངས་ལྡན། འདྲ་བ་ལྟ་བུའི་ཁྱད་པར། ཐ་དད་ལྟ་བུའི་གཅིག་པའོ། །དང་པོ་(སྡེ་ཚན་དུ་ངེས་པའི་རབ་དབྱེ་)ལ་གཉིས། བསྟན། བཤད་དོ། །དང་པོ་(མདོར་བསྟན་པ་)ནི། སེམས་དང་སེམས་བྱུང་ངེས་ལྷན་ཅིག །ཅེས་བཤད་ན། སེམས་བྱུང་དེ་ལ་དུ་ཡོད་ཅེ་ན། སྡེ་ཚན་དུ་ངེས་པའི་སེམ༷ས་བྱུང༷་རྣམ༷་པ་ལྔ༷་ཡོད་དེ༷། སེམས་ཀྱི་ས༷་མང༷་པོ་བ་དང༌། ལ༷་སོག༷ས་པ་དགེ་བའི་ས་མང་དང༌། ཉོན་མོངས་ཆེན་པོའི་ས་མང་དང༌། མི་དགེ་བའི་ས་མང་དང༌། ཉོན་མོངས་ཆུང་ངུའི་ས་མང་ལ་སོགས་པ་ཐ༷་དད༷་པ་རྣམས་སུ་ཡོད་པའི་ཕྱིར༷། ། གཉིས་པ་རྒྱས་བཤད་ལ་ལྔ་ལས། དང་པོ་ནི། སེམས་བྱུང་རྣམ་ལྔ་སྟེ། ས་མང་ལ་སོགས་ཐ་དད་ཕྱིར། །ཞེས་བཤད་ན། སེམས་ཀྱི་ས་མང་པོ་བ་དེ་ལ་དུ་ཡོད་ཅེ་ན། ཚོར༷་བ་དང༷༌། ཡུལ་ལ་གཡོ་བར་བྱེད་པའི་སེམ༷ས་པ༷་དང༌། འདུ༷་ཤེས༷་དང༷༌། ཡུལ་དོན་གཉེར་གྱི་འདུན༷་པ་དང༷༌། ཡུལ་དབང་རྣམ་ཤེས་གསུམ་འདུས་ནས་ཡུལ་ལ་ལོངས་སྤྱོད་པའི་རེག༷་པ་བྱ་བ་དང༌། ཆོས་རབ་ཏུ་རྣམ་འབྱེད་ཀྱི་བློ༷་གྲོས༷་དང༌། དམིགས་པ་མ་བརྗེད་པའི་དྲ༷ན་པ་དང༌། དམིགས་པ་བྱེ་བྲག་པ་ལ་སེམས་གཏོད་པའི་ཡིད༷་ལ༷་བྱེད༷་པ་དང༷༌། དམིགས་པ་ལ་ཡོན་ཏན་དུ་འཛིན་པའི་མོས༷་པ༷་དང༷༌། སེམས་རྩེ་གཅིག་པའི་ཏིང༷་ངེ༷་འཛི༷ན་རྣམས་ནི་སེམ༷ས་ཀྱི་ས་མང་པོ་བ་ཡིན་ཏེ། སེམས་ཐམ༷ས་ཅ༷ད་ལ༷་འཁོར་དུ་འབྱུང་བའི་སེམས་བྱུང་ཡིན་པའི་ཕྱིར། གཉིས་པ་དགེ་བའི་ས་མང་པོ་བ་ནི། སེམས་ཉོན་མོངས་པ་དང༌། ཉེ་བའི་ཉོན་མོངས་པ་ལས་དང་བར་བྱེད་པའི་དད༷་པ་དང༷༌། ཡོན་ཏན་ཅན་ལ་གཅེས་སྤྲས་སུ་བྱེད་པའི་བག༷་ཡོད༷་དང༌། སེམས་ལས་སུ་རུང་བའི་ཤིན༷་ཏུ༷་སྦྱང༷ས་པ་དང༌། སེམས་བྱིང་རྒོད་ཀྱི་དབང་དུ་མི་འགྲོ་བར་ལྷུན་གྲུབ་ཏུ་འབྱུང་བའི་བཏང༷་སྙོམ༷ས་དང༌། ཡོན་ཏན་དང་ཡོན་ཏན་ཅན་ལ་གུས་པའི་ངོ༷་ཚ༷་ཤེས༷་པ་དང༌། ཁ་ན་མ་ཐོ་བར་འཇིག་པར་ལྟ་བའི་ཁྲེལ༷་ཡོད༷་པ་དང༌། རྩ༷་བ༷་གཉིས༷་ནི་འདོད་ཆགས་མེད་པ་དང་ཞེ་སྡང་མེད་པ་ཡིན་ལ། དེ་གཉིས་ནི་རང་ཡུལ་ལ་རང་སྟོབས་ཀྱིས་ཆགས་པ་མེད་པ་དང༌། སྡང་བ་མེད་པའི་རྣམ༷་པར་ཞུགས་པའི་བློའོ། །གཞན་ལ་མཐོ་འཚམས་པའི་རྣམ་པར་མ་ཞུགས་པའི་རྣམ་པར་མི༷་འཚེ༷་བ་དང༷༌། དགེ་བ་ལ་སེམས་སྤྲོ་བའི་བརྩོ༷ན་འགྲུ༷ས་རྣམས་ནི་དགེ་བའི་ས་མང་པོ་བ་ཡིན་ཏེ། རྟག༷་ཏུ༷་དགེ༷་བའི༷་སེམས་ཐམས་ཅད་ཀྱི་འཁོར་དུ་འབྱུང༷་བའི་ཕྱིར། གསུམ་པ་ཉོན་མོངས་ཆེན་པོའི་ས་པ་ནི། ལས་འབྲས་སོགས་མི་ཤེས་པའི་རྨོང༷ས་པ་མ་རིག་པ་དང༷༌། བག་ཡོད་ཀྱི་མི་མཐུན་ཕྱོགས་བག༷་མེད༷་དང༌། བརྩོན་འགྲུས་ཀྱི་མི་མཐུན་ཕྱོགས་ལེ༷་ལོ༷་དང༷༌། དད་པའི་མི་མཐུན་ཕྱོགས་མ༷་དད༷་པ༷་དང༷༌། སེམས་བློང་བློང་པོར་བྱེད་པའི་རྨུག༷ས་པ་དང༷༌། ཕྱིར་འཕྲོ་བའི་རྒོད༷་པ་རྣམས་ནི་ཉོན་མོངས་ཆེན་པོའི་ས་པ་ཡིན་ཏེ། རྟག་ཏུ་ཉོན༷་མོང༷ས་ཅན༷་གྱི་སེམས་ཐམས་ཅད་ལ༷་འཁོར་དུ་འགྱུར་བའི་ཕྱིར། བཞི་པ་མི༷་དགེ༷་བའི་ས་མང་པོ་བ་ནི༷། ཁྲེལ༷་མེད༷་པ་དང༷་ངོ༷་ཚ༷་མེད༷་པ༷་གཉིས་ནི་མི་དགེ་བའི་ས་མང་པོ་བ་ཡིན་ཏེ། མི་དགེ་བའི་སེམས་ཐམས་ཅད་ཀྱི་འཁོར་དུ་འབྱུང་གི །གཞན་གྱི་འཁོར་དུ་མི་འབྱུང་བའི་ཕྱིར། ལྔ་པ་ཉོན་མོངས་ཆུང་ངུའི་ས་པ་ནི། གནོད་བྱེད་མཐོང་བས་སེམས་ཁོང་ནས་འཁྲུག་པའམ་མནར་སེམས་པའི་ཁྲོ༷་བ༷་དང༷༌། ཁྲོས་པའི་རྗེས་སུ་ཀུན་ནས་མནར་སེམས་མང་དུ་བྱེད་པའི་ཁོན༷་དུ༷་འཛི༷ན་པ་དང༷༌། སེམས་དྲང་པོ་མིན་པའི་གྱ་གྱུར་བྱེད་པའི་གཡོ༷་དང༷་ནི༷། གཞན་གྱི་ཕུན་ཚོགས་ལ་མི་བཟོད་པའི་ཕྲག༷་དོག༷་དང༌། སྡིག་པ་ལ་དམ་དུ་འཛིན་པའི་འཚི༷ག་པ་དང༌། ཉེས་པ་སྦེད་པར་བྱེད་པའི་འཆ༷བ་པ་དང༌། ཆོས་དང་ཟང་ཟིང་སོགས་གཞན་ལ་མི་གཏོང་ཞིང་གཏོང་བ་དང་འགལ་བའི་རྣམ་པར་ཞུགས་པའི་སེ༷ར་སྣ༷་དང༷༌། ཡོན་ཏན་མེད་བཞིན་དུ་ཡོན་ཏན་དང་ལྡན་པར་བཅོས་ནས་གཞན་འབྲིད་པར་བྱེད་པའི་སྒྱུ༷་དང༷༌། རྒྱག༷ས་པ་དང༷༌། གཞན་ལ་མཐོ་འཚམས་པའི་རྣམ༷་འཚེ༷་རྣམས་ནི༷་ཉོན༷་མོང༷ས་ཆུང༷་ངུ༷འི་ས༷་པ༷་རྣམ༷ས་ཡིན་ཏེ། སྤང་བྱ་རིས་ཀྱིས་ཆུང༌། མཚུངས་ལྡན་གྲོགས་ཀྱིས་ཆུང། འབྱུང་ས་རྟེན་གྱིས་ཆུང་བའི་སེམས་བྱུང་ཡིན་པའི་ཕྱིར། དང་པོ་གྲུབ་སྟེ། སྒོམ་སྤང་ཙམ་ཡིན་པའི་ཕྱིར། གཉིས་པ་གྲུབ་སྟེ། ཡིད་ཀྱི་རྣམ་ཤེས་ཙམ་དང་མཚུངས་ལྡན་ཡིན་པའི་ཕྱིར། གསུམ་པ་གྲུབ་སྟེ། ཡིད་ཙམ་ལ་བརྟེན་པ་ཡིན་པའི་ཕྱིར།
