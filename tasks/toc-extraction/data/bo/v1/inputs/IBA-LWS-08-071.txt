གཉིས་པ་མྱོང་འགྱུར་གྱི་དབྱེ་བ་ལ་གཉིས། ལྔར་དབྱེ་བ། རྣམ་སྨིན་མྱོང་འགྱུར་བྱེ་བྲག་ཏུ་བཤད་པའོ། །དང་པོ་(ལྔར་དབྱེ་བ་)ནི། མྱོང་བར་འགྱུར་བའི་ལས་རྣམ་པ་ལྔ་ཡོད་པ་ཡིན་ཏེ། ཚོར་བ་ལྟ་བུ་ངོ༷་བོ༷་ཉིད༷་ཀྱིས་ཏེ་རང་གི་མཚན་ཉིད་ཀྱིས་མྱོང་བར་འགྱུར་བ་དང༷༌། རེག་པ་ལྟ་བུ་མཚུང༷ས་ལྡན༷་གྱིས་མྱོང་བར་འགྱུར་བ་དང༷༌། གཟུགས་ལྟ་བུ་དམིགས༷་པས༷་མྱོང་བར་འགྱུར་བ་དང༷༌། དགེ་མི་དགེའི་ལས་ལྟ་བུ་རྣམ༷་སྨིན༷་གྱིས་བདེ་བ་སོགས་མྱོང་བར་འགྱུར་བ་དང༷། ཚོར་བ་བདེ་བ་ལྟ་བུ་མངོན༷་སུམ༷་དུ༷་གྱུར༷་པ༷་ལས༷་ཚོར་བ་གཞན་གཉིས་མྱོང༷་བར་འགྱུར༷་བ་མེད་པ་སྟེ། དེ་ལྟ་ན་ལས་ནི་རྣམ༷་པ༷་ལྔ༷་རུ་ཡོད་པའི་ཕྱིར། གཉིས་པ་རྣམ་སྨིན་མྱོང་འགྱུར་བྱེ་བྲག་ཏུ་བཤད་པ་ལ་གཉིས། སྤྱིར་བསྟན། མྱོང་ངེས་བྱེ་བྲག་ཏུ་བཤད་པའོ། །དང་པོ་(སྤྱིར་བསྟན་)ལ་གཉིས། ལས་ཀྱི་དབྱེ་བ། ཁམས་དང་འགྲོ་བ་གང་ན་ལས་འཕན་པ་དུ་ཡོད་གཉིས། དང་པོ་(ལས་ཀྱི་དབྱེ་བ་)ནི། རྣམ་སྨིན་མྱོང་འགྱུར་གྱི་ལས་དེ་ལ་བཞི་སྟེ། དེ༷་ལ་ཡང༷་ངེས༷་པ་དང༷༌། མ༷་ངེས༷་པའི་ལས་གཉིས་ལས། དང་པོ་ལ། མཐོང༷་བའི༷་ཆོས༷་སྐྱེ་བ་འདི་ཉིད་ལ༷་མྱོང་བ་དང༌། སོག༷ས་པ༷་སྐྱེ་བ་ཕྱི་མར་མྱོང་བ་སྐྱེ་ནས་མྱོང་འགྱུར་དང༌། དེ་ཕན་ཆད་ལ༷་མྱོང་བ་ལན་གྲངས་གཞན་མྱོང༷་འགྱུར༷་གསུམ་ཡོད་པའི་ཕྱིར༷་ན༷་མྱོང་བར་ངེས༷་པ་རྣམ༷་པ་གསུམ༷་དང༌། མ་ངེས་པ་སྟེ་བཞི་ཡིན་ནོ། །སྔོན་གྱི་སློབ་དཔོན་ཁ༷་ཅིག༷་ནི་མྱོང་བ་མ་ངེས་པ་ལའང་རྣམ་པར་སྨིན་པ་ངེས་ལ་གནས་སྐབས་མ་ངེས་པ་དང༌། དེ་གཉིས་ཀ་མ་ངེས་པ་གཉིས་སུ་ཕྱེ་ནས་ལས༷་རྣམ༷་པ་ལྔ༷་ཡིན་ནོ་ཞེས༷་ཟེར༷། དཔེ་སྟོན་པའི་སྡེ་པ་གཞན༷་དག༷་ནི་མུ༷་ནི༷་བཞི༷་ཡིན་ནོ་ཞེས༷་ཟེར༷་རོ། །དེ་ལ་གནས་སྐབས་ངེས་ལ་རྣམ་སྨིན་མྱོང་བར་མ་ངེས་པ་ནི་ལས་འགའ་ཞིག་འབྲས་བུ་མྱོང་ན་མཐོང་ཆོས་སོགས་གང་རུང་དུ་མྱོང་ལ། ནུས་པ་ཆུང་བས་རྣམ་སྨིན་མྱོང་བར་མ་ངེས་པ་དང༌། རྣམ་སྨིན་ངེས་ལ་གནས་སྐབས་མ་ངེས་པ་ནི་རྣམ་སྨིན་ནི་གདོན་མི་ཟ་བར་མྱོང་ལ། མཐོང་ཆོས་སོགས་གནས་སྐབས་གསུམ་པོ་མ་ངེས་པའོ། །གཉིས་ཀ་ངེས་པ་ནི་མཐོང་ཆོས་ལ་རྣམ་སྨིན་མྱོང་བར་ངེས་པའོ། །གཉིས་ཀ་མ་ངེས་པ་ནི་གནས་སྐབས་དེར་མ་ངེས་ཤིང་རྣམ་སྨིན་མྱོང་བའང་མ་ངེས་པའོ། །ལས་གང་གིས་ཕུང་པོའི་རྒྱུན་ནམ་རིས་མཐུན་འཕེན་ཅེ་ན། སྐྱེས་ནས་མྱོང་བ་ལ་སོགས་པ་གསུམ༷་གྱིས༷་ནི་སྐྱེ་བ་གཞན་རིས༷་མཐུན་པའི་ཕུང་པོའི་རྒྱུན་འཕེན༷་པར༷་བྱེད༷་ཀྱི། མཐོང་ཆོས་ལ་མྱོང་འགྱུར་གྱི་ལས་ཀྱིས་རིས་མཐུན་མི་འཕེན་ཏེ། ལས་བྱེད་པའི་ཕུང་པོའི་རྒྱུན་དེ་ཉིད་ལ་རྣམ་པར་སྨིན་པའི་ཕྱིར། ། གཉིས་པ་(ཁམས་དང་འགྲོ་བ་གང་ན་ལས་འཕན་པ་དུ་ཡོད་གཉིས་)ནི། ཁམས་དང་འགྲོ་བ་གང་ན་ལས་ཀྱི་འཕེན་པ་དུ་ཡོད་ཅེ་ན། འགྲོ་བ་ལྔ་པོ་ཐམ༷ས་ཅད༷་ན༷་ནི༷་ལས་ཀྱི་འཕེན༷་པ༷་བཞི༷་ཡོད། འོན་ཀྱང་དམྱལ༷་བ༷་ན༷་ནི༷་དགེ༷་བའི༷་ལས་ཀྱི་འཕེན་པ་སྐྱེས་ནས་མྱོང་འགྱུར་སོགས་གསུམ༷་ཡོད་ཀྱི། མཐོང་ཆོས་ལ་མྱོང་འགྱུར་གྱི་དགེ་བའི་ལས་འཕེན་མེད་དེ། དེ་ན་རྣམ་སྨིན་ཡིད་དུ་འོང་བ་མེད་པའི་ཕྱིར། སྲིད་རྩེ་མ་གཏོགས་པ་བརྒྱད་པོ་གང༷་ལས༷་འདོད༷་ཆག༷ས་དང་བྲལ༷་བ་བརྟན༷་པའི༷་བྱིས༷་པ༷་འདོད་ཆགས་དང་བྲལ་བའི་ས་དེར༷་སྐྱེས༷་ནས་མྱོང༷་འགྱུར་གྱི་ལས་མི༷་བྱེད༷་དེ། དེ་ཕྱི་མ་ལ་འདོད་ཆགས་དང་བྲལ་བའི་ས་དེར་མི་སྐྱེ་བའི་ཕྱིར། ས་བར་པ་བདུན་ལས་འདོད་ཆགས་དང་བྲལ་བ་བརྟན་པའི་འཕགས་པས་དེར་སྐྱེས་ནས་མྱོང་འགྱུར་དུ་མ་ཟད། ལན་གྲངས་གཞན༷་དུ༷འང༷་མྱོང༷་འགྱུར་གྱི་ལས་མི༷་བྱེད༷་དེ། དེ་འདོད་ཆགས་དང་བྲལ་བའི་ས་དེར་གཏན་ནས་མི་སྐྱེ་བའི་ཕྱིར། འདོད༷་པ་དང་སྲིད་རྩེ་ལས་འདོད་ཆགས་དང་བྲལ་བའི་འཕགས་པ་མི༷་བརྟན༷་པ༷་དེ་ཡང༷་འདོད་རྩེར་སྐྱེས་ནས་མྱོང་འགྱུར་དང༌། ལན་གྲངས་གཞན་ལ་མྱོང་འགྱུར་གྱི་ལས་བྱེད་པ་མིན༷་ཏེ། འདོད་པ་དང་སྲིད་རྩེ་ལ་འདོད་ཆགས་དང་བྲལ་བ་ལས་ཉམས་པའི་འཕགས་པ་ནི་འབྲས་བུ་ལས་ཉམས་པ་ཡིན་ལ། དེ་ལས་ཉམས་པ་སོར་མ་ཆུད་ཀྱི་བར་དུ་འཆི་བ་མི་སྲིད་པའི་ཕྱིར། བར་དོ་ནའང་ལས་ཀྱི་འཕེན་པ་ཡོད་དམ་ཞེ་ན། ཡོད་དེ། སྲིད༷་པ༷་བར༷་མ༷་འདོད༷་ཁམ༷ས་སུ༷་མངལ་སྒོང་ལ་ཉིང་མཚམས་སྦྱོར་ན་བར་དོ་དེ་ཉིད་དང༌། ནུར་ནུར་པོ་སོགས་མངལ་གྱི་གནས་སྐབས་ལྔ། བྱིས་པ་སོགས་བཙས་པའི་གནས་སྐབས་ལྔ་སྟེ་བཅུ་གཅིག་མྱོང་བར་ངེས་མ་ངེས་གཉིས་གཉིས་ཏེ་འཕེན༷་པ༷་རྣམ༷་པ༷་ཉི༷་ཤུ༷་རྩ་གཉིས༷་ཡོད་ལ། ངེས་པ་བཅུ་གཅིག་པོ་དེ༷་ནི༷་མཐོང༷་ཆོས༷་ལ་འབྲས༷་བུ༷་ཅན་གྱི་ལས་བྱེད་པ་ཡིན༷་ཏེ། དེ༷་ནི༷་ལས་སོག་པའི་གནས་སྐབས་ཀྱི་ཕུང་པོ་དང༌། རྣམ་སྨིན་མྱོང་བའི་གནས་སྐབས་ཀྱི་ཕུང་པོ་གཉིས་འཕེན་བྱེད་ཀྱི་ལས་གཅིག་གིས་འཕངས་པའི་རིས༷་མཐུན་གཅིག༷་ཁོ༷་ན༷་ཡིན༷་པའི་ཕྱིར། ། གཉིས་པ་མྱོང་ངེས་བྱེ་བྲག་ཏུ་བཤད་པ་ནི། རྣམ་སྨིན་མྱོང་ངེས་ཀྱི་ལས་དེ་གང་ཞེ་ན། ཉོན༷་མོངས༷་པ་དང༌། རབ༷་ཏུ་དང༷་བ་ཤུགས་དྲག༷་པོ༷་དང༷༌། ཡོན༷་ཏན༷་གྱི་ཞིང༷་འོག་ནས་འཆད་པར་ལྟར་དཀོན་མཆོག་སོགས་ལ་བརྟེན་པ་དང༷༌། སྦྱོར་བ་རྒྱུན༷་ཆག༷ས་སུ༷་བྱས༷་པ༷་གང༷་ཞིག༷་ཡིན༷་པ༷་དང༷༌། པར་སིག་པ་ལྟར་བསོད་ནམས་ཀྱི་བློས་ཀྱང་རུང་ཕ༷་མ༷་གསོད༷་པའི་ལས་གང༷་ཡིན་པ་དེ༷་རྣམ་སྨིན་མྱོང་བར་ངེས༷་པའོ༷། །དེ་ནི་མཐོང་ཆོས་འབྲས་བུ་ཡིན། །ཞེས་བཤད་ན། དེ་གང་ཞེ་ན། མཐོང༷་བའི་ཆོས༷་ལ་འབྲས༷་བུ༷་མྱོང་བ་ཅན༷་གྱི༷་ལས༷་ནི་ཞིང༷་དང༷་བསམ༷་པའི༷་ཁྱད༷་པར༷་ལས༷་འགྱུར་བ་ཡིན་ཏེ། ཞིང་གི་ཁྱད་པར་ལས་མཐོང་ཆོས་འབྲས་བུ་ཅན་གྱི་ལས་སུ་འགྱུར་བ་ནི་སྔོན་བདག་ཅག་གི་སྟོན་པ་དགེ་སློང་སྡེ་སྣོད་གསུམ་པར་གྱུར་པ་ན། དགེ་འདུན་གྱིས་རྩོད་པ་འཕེལ་བར་བྱས་པ་དགེ་སློང་སྡེ་སྣོད་འཛིན་པ་ཞིག་གིས་བསྡུམས་པས། དགེ་སློང་སྡེ་སྣོད་གསུམ་པ་དེ་ཁྲོས་ནས་བུད་མེད་ཀྱིས་བུད་མེད་ཀྱི་རྩོད་པ་ཞི་བར་བྱས་སོ་ཞེས་སྐུར་པ་བཏབ་པ་ལས་ཚེ་དེ་ཉིད་ལ་བུད་མེད་དུ་གྱུར་ཅིང༌། དེ་ནས་སྐྱེ་བ་ལྔ་བརྒྱར་བུད་མེད་དུ་གྱུར་པ་ལྟ་བུའོ། །བསམ་པའི་ཁྱད་པར་ལས་མཐོང་ཆོས་འབྲས་བུ་ཅན་གྱི་ལས་སུ་འགྱུར་བ་ནི་རྒྱལ་པོ་ཀ་ན་ཀའི་ཉུག་རུམ་པ་ལོང་བ་ཞེས་པས། མི་ཞིག་གིས་གླང་ལྔ་བརྒྱ་འདེད་པ་ལ་ཅི་བྱེད་དྲིས་པས། འབྲས་བུ་འབྱིན་པ་ཡིན་ཟེར་བ་ལ། སྙིང་རྗེས་དེ་དག་དེ་ལས་ཐར་བར་བྱས་པས་ཚེ་དེ་ཉིད་ལ་མཚན་སོར་ཆུད་པ་ལྟ་བུའོ། །དེར་མ་ཟད་ས༷་དེའི༷་འདོད༷་ཆག༷ས་དང་གཏན༷་དུ་བྲལ༷་བའི་ཕྱིར༷། ས་དེར་རྣམ༷་སྨིན༷་ངེས༷་པ༷་གང༷་ཡིན༷་པ༷འང༷་མཐོང་ཆོས་འབྲས་བུ་ཅན་གྱི་ལས་ཡིན་ཏེ། ཡུལ་འཕོ་བ་ན་བུ་ལོན་ལྡང་བ་བཞིན་ནོ། །ཞིང་གི་ཁྱད་པར་ཅི་ལྟ་བུ་ཞེ་ན། གང༷་དག༷་འགོག༷་པ་ལས་ལངས་མ་ཐག་པ་དེ་ལ་ཕན་གནོད་བྱས་པས་འབྲས་བུ་འཕྲལ་དུ་མྱོང་བར་འགྱུར་ཏེ། དེ་མྱ་ངན་ལས་འདས་པ་སླར་ལོག་པ་དང་འདྲ་བའི་ཕྱིར། བྱམ༷ས་པ་ཚད་མེད་དང་ཉོན༷་མོང༷ས་མེད༷་པའི་ཏིང་ངེ་འཛིན་ལས་ལངས་མ་ཐག་པ་དེ་ལ་ཕན་གནོད་བྱས་པས་འབྲས་བུ་འཕྲལ་དུ་མྱོང་བར་འགྱུར་ཏེ། སེམས་ཅན་ཐམས་ཅད་ལ་ཕན་པ་དང་ཉོན་མོངས་པ་མེད་པའི་ལྷག་པའི་བསམ་པ་དང་འབྲེལ་ཞིང༌། རྣོ་ཞིང་གསལ་བའི་བསོད་ནམས་ཚད་མེད་པས་ཡོངས་སུ་བསྒོས་པའི་ཕྱིར། མཐོང་ལམ་ལས་ལངས་མ་ཐག་པ་དེ་ལ་ཕན་གནོད་བྱས་པས་འབྲས་བུ་འཕྲལ་དུ་མྱོང་བར་འགྱུར་ཏེ། དེས་མཐོང༷་སྤང་མ་ལུས་པར་སྤངས་པའི་གནས་གྱུར་གསར་པ་དྲི་མ་མེད་པ་ཐོབ་པའི་ཕྱིར། དགྲ༷་བཅོམ༷་པའི་འབྲས༷་བུ་ལས་ལང༷ས་མ་ཐག་པ༷་དེ༷་ལ༷་ཕན༷་གནོད༷་བྱས༷་པས༷་འབྲས༷་བུ༷་འཕྲལ༷་དུ༷་མྱོང༷་བར༷་འགྱུར༷་ཏེ། སྒོམ་སྤང་མ་ལུས་པར་སྤངས་པའི་གནས་གྱུར་གསར་པ་དྲི་མ་མེད་པ་ཐོབ་པའི་ཕྱིར། ཕྱིར་འོང་དང་ཕྱིར་མི་འོང་གི་འབྲས་བུ་ལས་ལངས་མ་ཐག་པ་དེ་ལ་ཕན་གནོད་བྱས་པས་འབྲས་བུ་འཕྲལ་དུ་མྱོང་བར་མི་འགྱུར་ཏེ། མཐོང་སྤང་སྤངས་པའི་གནས་གྱུར་གསར་པ་དྲི་མ་མེད་པ་ནི་རྙིང༌། སྒོམ་སྤང་མ་ལུས་པར་སྤངས་པའི་གནས་གྱུར་གསར་པ་དྲི་མ་མེད་པ་མ་ཐོབ་པའི་ཕྱིར། །
