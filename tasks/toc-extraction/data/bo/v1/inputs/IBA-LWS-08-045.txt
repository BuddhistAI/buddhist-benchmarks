གསུམ་པ་བསྡུ་བ་ནི། སྲིད་པ་བཞི་བཤད་ན། དེ་དག་གི་ངོ་བོ་གང་ཞེ་ན། སྲིད༷་པ༷་བཞི༷་པོ༷་དེ༷་དག༷་ལ༷ས་སྐྱེ༷་བའི༷་སྲིད༷་པ༷་ནི་ཉོན༷་མོང༷ས་ཅ༷ན་ནོ། །འོ་ན། ཉོན་མོངས་པ་གང་གིས་སྐྱེ་སྲིད་དུ་ཉིང་མཚམས་སྦྱོར་ཞེ་ན། རང༷་གི༷་ས༷་པ༷འི་རྩ་བའི་ཉོན༷་མོང༷ས་པ་ཀུན༷་གྱིས་སྦྱོར་གྱི། གཞ༷ན་གྱི་ས་པའི་ཉོན་མོངས་པས་ནི་མིན་ནོ། །སྐྱེ་སྲིད་ལས་གཞན་བར་སྲིད་དང༌། སྔོན་དུས་ཀྱི་སྲིད་པ་དང༌། འཆི་སྲིད་གསུམ༷་ནི་དགེ་མི་དགེ་ལུང་མ་བསྟན་རྣམ་གསུམ་མོ། །དེ་དག་ཁམས་གང་ན་དུ་ཡོད་ཅེ་ན། འདོད་གཟུགས་གཉིས་ན་བཞི་ག་ཡོད་ལ། གཟུག༷ས་མེ༷ད་ན་ནི་གསུམ༷་ཡིན་ཏེ། བར་སྲིད་མེད་པའི་ཕྱིར་རོ།། །།
