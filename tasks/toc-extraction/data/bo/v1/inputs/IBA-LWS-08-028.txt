ལྔ་པ་སྲོག་ནི། སྲོག་དང༌། ཞེས་བཤད་ན། སྲོག་དེ་གང་ཞེ་ན། སྲོག༷་ནི༷་ཚེ༷་ཡི༷ན་ཏེ། གདགས་པའི་བསྟན་བཅོས་ལས། སྲོག་གང་ཞེ་ན། ཁམས་གསུམ་པའི་ཚེའོ། །ཞེས་གསུངས་པའི་ཕྱིར། ཚེ་གང་ཞེ་ན། ཚེ་ནི་ལུས་ཀྱི་དྲོད༷་དང༷་རྣམ༷་པར་ཤེས༷་པ་གནས་པའི་རྟེན༷་བྱེད་པའི་མི་ལྡན་པའི་འདུ་བྱེད་རྫས་གྲུབ་གང༷་ཡི༷ན་པ༷འོ༷། །དེའང་འདོད་པ་དང་གཟུགས་སུ་དྲོད་དང་རྣམ་ཤེས་གཉིས་ཀའི་རྟེན་ཡིན་ལ། གཟུགས་མེད་ཁམས་སུ་ནི་རྣམ་ཤེས་ཁོ་ནའི་རྟེན་ཡིན་ནོ། འོ་ན། ཚེ་ཟད་པ་ཁོ་ནས་འཆིའམ་ཞེ་ན། འདི་ལ་མུ་བཞི་ཡོད་དེ། ཚེ་ཟད་ལ་བསོད་ནམས་མ་ཟད་པར་འཆི་བ་ནི་ཚེ་རྣམ་པར་སྨིན་པའི་ལས་ཟད། ལོངས་སྤྱོད་རྣམ་པར་སྨིན་པར་འགྱུར་བའི་ལས་མ་ཟད་པ་ལྟ་བུའོ། །ཚེ་མ་ཟད་ལ་བསོད་ནམས་ཟད་ནས་འཆི་བ་ནི་ཚེ་རྣམ་སྨིན་གྱི་ལས་མ་ཟད་པར། ལོངས་སྤྱོད་རྣམ་སྨིན་གྱི་ལས་ཟད་ནས་འཆི་བ་ལྟ་བུའོ། །གཉིས་ཀ་ཡིན་པ་ནི་ཚེ་དང་བསོད་ནམས་རྣམ་སྨིན་གྱི་ལས་གཉིས་ཀ་ཟད་ནས་འཆི་བ་ལྟ་བུའོ། །གཉིས་ཀ་མ་ཡིན་པ་ནི་ཚེ་རྣམ་པར་སྨིན་པར་འགྱུར་བའི་ལས་དང༌། ལོངས་སྤྱོད་རྣམ་པར་སྨིན་པར་འགྱུར་བའི་ལས་གཉིས་ཀ་མ་ཟད་པར་དུས་མ་ཡིན་པར་འཆི་བ་ལྟ་བུའོ། ། དྲུག་པ་མཚན་ཉིད་ནི། མཚ༷ན་ཉིད༷་དག༷་ནི༷་སྐྱེ༷་བ༷་དང༷༌། རྒ༷་བ་དང༷༌། གན༷ས་པ་དང༷༌། མི༷་རྟག༷་པ་ཉིད༷་དོ། །དེའང་ཆོས་དེ་འདུས་བྱས་སུ་མཚོན་པས་ན་མཚན་ཉིད། སྐྱེ་བ་ནི་སྐྱེ་བར་བྱེད་པ་དང༌། རྒ་བ་ནི་རྒས་པར་བྱེད་པ་དང༌། གནས་པ་ནི་གནས་པར་བྱེད་པ་དང༌། མི་རྟག་པ་ནི་འཇིག་པར་བྱེད་པའོ། །དེ་ལ་སྐྱེ་བ་ལ་སོགས་པའང་འདུས་བྱས་ཡིན་པས་སྐྱེ་བ་སོགས་གཞན་ཡོད་དམ་ཞེ་ན། ཡོད་དེ། སྐྱེ༷་བ་སོགས་དེ༷་དག༷་ལའང་སྐྱེ༷་བའི༷་སྐྱེ༷་བ་ལ༷་སོག༷ས་པ་ཡོད་པའི་ཕྱིར། སྐྱེ་བའི་སྐྱེ་བ་སོགས་ཀྱང་འདུས་བྱས་ཡིན་པས་སྐྱེ་བ་སོགས་ཡོད་དགོས་ལ། དེ་ལྟ་ན་ཐུག་མེད་དུ་འགྱུར་རོ་ཞེ་ན། མི་འགྱུར་ཏེ། བུམ་པ་ལྟ་བུ་ཞིག་སྐྱེ་བའི་ཚེ་ན། རང་དང་རྩ་བའི་མཚན་ཉིད་སྐྱེ་སོགས་བཞི་དང༌། རྗེས་མཐུན་གྱི་མཚན་ཉིད་རྩ་བའི་སྐྱེ་བ་སྐྱེད་པར་བྱེད་པའི་སྐྱེ་བའི་སྐྱེ་བ་སོགས་བཞི་སྟེ་དགུ་མ་འོངས་པ་ནས་ད་ལྟར་དུ་ལྷན་ཅིག་སྐྱེ་ལ། དེ༷འི་ཚེ་སྐྱེ་བ་དེས་རང་མ་གཏོགས་པའི་ཆོས༷་བརྒྱད༷་སྐྱེད་ལ། སྐྱེ་བའི་སྐྱེ་བ་དེས་སྐྱེ་བ་གཅི༷ག་པུ་ལ༷་འཇུག༷་ཅིང་སྐྱེད་པའི་ཕྱིར། འོ་ན། འདུ་བྱེད་ཀྱི་རྒྱུན་ལ་སྐྱེ་བ་སོགས་ཇི་ལྟར་ཡོད་ཅེ་ན། ཡོད་དེ། རྒྱུན་གྱི་དང་པོ་སྐྱེ་བ་དང༌། རྒྱུན་གྱི་རྗེས་སུ་མཐུན་པར་འཇུག་པ་གནས་པ་དང༌། རྒྱུན་སྔ་ཕྱིའི་བྱེ་བྲག་རྒ་བ་དང༌། རྒྱུན་ལྡོག་པ་འཇིག་པ་ཡིན་པའི་ཕྱིར། ཞེས་ཟེར་རོ། །འདིའང་བྱེ་སྨྲའི་རྟོག་བཏགས་སོ། །སྐྱེ་བ་མ་འོངས་པས་ཆོས་དེ་སྐྱེད་ན། ཆོས་ཐམས་ཅད་ཅིག་ཆར་སྐྱེ་བར་འགྱུར་རོ་ཞེ་ན། རང་རང་གི་རྒྱུ༷་དང༷་རྐྱེན༷་དག༷་ཚོགས་པ་མེད༷་པར༷་ནི༷་སྐྱེ༷་བ༷་འབའ་ཞིག་གིས་བསྐྱེད༷་པར་བྱ༷་བ་མ་འོངས་པ་རྣམས་ད་ལྟར་དུ་སྐྱེད༷་པར་བྱེད༷་པ་མིན༷་ཏེ། རྒྱུ་རྐྱེན་ཚོགས་པ་ཚང་བས་ད་ལྟར་དུ་བསྐྱེད་པའི་ཕྱིར། འོ་ན། སྐྱེ་བ་དོན་མེད་པར་འགྱུར་རོ་ཞེ་ན། མིན་ཏེ། དབང་སོགས་བཞིན་ནོ། ། བདུན་པ་བརྗོད་བྱེད་ནི། སྔར་མདོར་བསྟན་པའི་མིང༷་གི༷་ཚོག༷ས་དང༌། ལ༷་སོག༷ས་པ༷་ཚིག་གི་ཚོགས་དང༌། ཡི་གེའི་ཚོགས་ནི༷་གང་ཡིན་ཞེ་ན། མིང༷་ནི་དོན་གྱི་ངོ་བོ་ཙམ་བརྗོད་པ་གཟུགས་ཞེས་པའམ་བུམ་པ་ཞེས་པ་ལྟ་བུ་སྟེ། ཡི་གེ་ག་དང༌། ཟ་དང༌། ཞབས་ཀྱུ་དང༌། ག་དང༌། ས་རྣམས་འདུས་པ་ན་མིག་གི་སྐྱེ་མཆེད་ཀྱི་ཡུལ་གོ་བར་བྱེད་པའི་མི་ལྡན་པའི་འདུ་བྱེད་རྫས་གྲུབ་ཅིག་ཡོད་དོ་ཞེས་དང༌། ངག༷་གམ་ཚིག་ནི་ཀྱེ་མ་འདུས་བྱས་རྣམས་མི་རྟག་ཅེས་པ་ལྟ་བུ་རྗོད་བྱེད་གང་གིས་སྐབས་ཀྱི་བརྗོད་བྱའི་དོན་རྫོགས་པར་སྟོན་པ་སྟེ། མིང་དུ་མ་འདུས་པ་ལ་བརྗོད་བྱའི་དོན་ཞིག་སྟོན་པའི་མི་ལྡན་པའི་རྫས་ཤིག་ལ་འདོད། ཡི༷་གེ༷་ནི་མིང་ཚིག་གི་གཞིར་གྱུར་པའམ་ཆ་ཤས་ཏེ། ཀ་ལ་སོགས་པའོ། །ངེས་ཚིག་ནི་ཨཀྵར་མི་འགྱུར་བ་སྟེ། ཡུལ་དུས་ཀྱི་བརྡའི་དབང་གིས་རང་གི་ངོ་བོ་གཞན་དུ་མི་འགྱུར་བའོ། །མཚན་གཞི་ནི་བློ་ངོའི་ཀ་ལ་སོགས་པའོ། །དེ་ནི་མིང་ཚིག་གི་གཞིར་གྱུར་པའི་མི་ལྡན་པའི་འདུ་བྱེད་ཀྱི་རྫས་ཤིག་ཡིན་ནོ། །ཚོག༷ས་ནི་ཕྲེང་བ་ལ་སོགས་པ་བཞིན་དུ་མང་པོ་འདུས་པའི་དོན་ཏོ། །དེ་གསུམ་ཁམས་གང་དུ་གཏོགས་ཤེ་ན། འདོ༷ད་ཁམས་དང༷་གཟུག༷ས་སུ་གཏོག༷ས་པ་ཡིན་གྱི། གཟུགས་མེད་དུ་མ་ཡིན་ཏེ། གཟུགས་མེད་ན་ངག་སོགས་མེད་པའི་ཕྱིར། སེམ༷ས་ཅན༷་དུ་སྟོན༷་པའང་ཡིན་ཏེ། རྩོལ་བ་ལས་བྱུང་བའི་ཕྱིར་དང༌། རྒྱུད་ཀྱིས་བསྡུས་པའི་ཕྱིར། དེ་དག་སྐལ་མཉམ་གྱི་རྒྱུས་བསྐྱེད་པས་བྱུང་བ་གསུམ་ལས། རྒྱུ༷་མཐུ༷ན་ལས་བྱུང་བ་ཡིན་གྱི། རྒྱས་བྱུང་དང་རྣམ་སྨིན་ལས་བྱུང་བ་ནི་མིན་ཏེ། རྡུལ་ཕྲན་བསགས་པ་མ་ཡིན་པས་རྒྱས་མི་ནུས་པ་དང༌། འདོད་རྒུར་འབྱུང་བའི་ཕྱིར། དེ་དག་དགེ་མི་དགེར་ལུང༷་བསྟན༷་པ་མིན༷་ཏེ། དགེ་རྩ་ཆད་པ་དང༌། ཉོན་མོངས་པ་སྤངས་པ་ལའང་ཡོད་པའི་ཕྱིར། དེས་ན་མ་བསྒྲིབས་ལུང་མ་བསྟན་ཡིན་ནོ། །ཇི་ལྟར་དེ་དག་སེམས་ཅན་དུ་སྟོན་པ་དང༌། རྒྱུ་མཐུན་ལས་བྱུང་བ་དང༌། མ་བསྒྲིབས་ལུང་མ་བསྟན་ཡིན་པ་དེ༷་བཞི༷ན་དུ་གོང་དུ་བཤད་པའི་སྐལ༷་མཉ༷མ་ཡང་ཡིན་ནོ། །དེར་མ་ཟད་སྐལ་མཉམ་འདི་ནི་རྣམ༷་པར༷་སྨིན༷་པ༷འང༷་ཡི༷ན་ཏེ། རྣམ་སྨིན་སྐྱེས་ཀྱི་མིག་སོགས་འདྲ་བར་བྱེད་པའི་རྫས་དེ་རྣམ་སྨིན་ཡིན་པའི་ཕྱིར། འདི་ཁམས༷་གསུ༷མ་དུ་གཏོགས་པ༷་ཡིན་ཏེ༷། སེམས་ཅན་ཐམས་ཅད་ལ་སེམས་ཅན་དུ་འདྲ་བར་བྱེད་པའི་རྫས་སམ་སྐལ་མཉམ་གྱི་རྫས་རེ་རེ་ཡོད་པའི་ཕྱིར། ཐོབ་པའང་རྒྱུ་མཐུན་ལས་བྱུང་བ་དང་རྣམ་སྨིན་ལས་བྱུང་བ་རྣམ༷་པ་གཉི༷ས་ཀ་ཡོད་པ་ཡིན་ནོ། །ཕྱི་མ་དེ་ཡིན་ཏེ། རྣམ་སྨིན་སྐྱེས་ཀྱི་མིག་སོགས་ཀྱི་ཐོབ་པ་དེ་རྣམ་སྨིན་ཡིན་པའི་ཕྱིར། མཚན༷་ཉིད༷་རྣམ༷ས་ཀྱང༷་རྒྱུ་མཐུན་ལས་བྱུང་བ་དང་རྣམ་སྨིན་ལས་བྱུང་བ་ཡིན་ནོ། །སྙོམ༷ས་འཇུ༷ག་གཉིས་དང༷་མི༷་ལྡན༷་པ༷་མ་ཐོབ་པ་དག༷་རྒྱུ༷་མཐུན༷་པ༷་ལས་བྱུང་བ་ཡིན་གྱི། རྣམ་སྨིན་ལས་བྱུང་བ་མ་ཡིན་ཏེ། སྙོམས་འཇུག་གཉིས་དགེ་བར་ངེས་ཤིང༌། མི་ལྡན་པ་དེ་ཞར་བྱུང་ཡིན་གྱི། བློ་སྔོན་དུ་བཏང་བའི་ལས་ཀྱིས་མ་བྱས་པའི་ཕྱིར། །
