གཉིས་པ་བླ་མ་བསྒོམ་པ་ལ། རྗེ་རིན་པོ་ཆེ་ཞལ་ནས། བླ་མ་སྐུ་བཞིའི་གངས་རི་ལ། ། མོས་གུས་ཀྱི་ཉི་མ་མ་ཤར་ན། ། བྱིན་རླབས་ཀྱི་ཆུ་རྒྱུན་མི་འབྱུང་བས། ། སེམས་མོས་གུས་འདི་ལ་ནན་ཏན་མཛོད། ། ཅེས་པ་འདི་ལ་གསུམ་སྟེ། སྤྲུལ་པའི་སྐུར་བསྒྲུབ་པ། ལོངས་སྤྱོད་རྫོགས་པའི་སྐུར་བསྒྲུབ་པ། ཆོས་ཀྱི་སྐུར་བསྒྲུབ་པའོ། །དང་པོ་ནི། དེ་ལྟར་ཡི་དམ་དུ་གསལ་བའི་སྤྱི་བོར། སེང་གེ་བརྒྱད་ཀྱིས་བཏེགས་པའི་ཁྲི་སྟེང་སྣ་ཚོགས་པདྨ་ཟླ་བའི་དཀྱིལ་འཁོར་གྱི་སྟེང་དུ་རང་གི་རྩ་བའི་བླ་ཉིད་བསྟན་པའི་བདག་པོ། སྐུ་མདོག་གསེར་བཙོ་མ་ལྟ་བུ་ཞབས་རྡོ་རྗེ་སྐྱིལ་ཀྲུང་གིས་བཞུགས་པ། ཕྱག་གཉིས་མཉམ་གཞག་དང་ས་མནོན། ན་བཟའ་གུར་ཀུམ་ཀུ་ཤ་རའི་མདོག་ལྟར་དམར་སེར་ཆོས་གོས་རྣམ་གསུམ་དང་ལྡན་པ། དབུའི་གཙུག་ཏོར་ལྟར་མི་མངོན་པ། མཚན་དང་དཔེ་བྱད་ཀྱིས་བརྒྱན་པ། འོད་དང་འོད་ཟེར་དཔག་ཏུ་མེད་པར་འཕྲོ་བར་བཞུགས་པ་ལ། རྩ་བརྒྱུད་ཀྱི་བླ་མ་དང་སངས་རྒྱས་བྱང་ཆུབ་སེམས་དཔའ་ཡི་དམ་ལྷ་ཚོགས་རྣམས་ཀྱིས་བསྐོར་ནས་བཞུགས་པར་བསམ་ལ། ཕྱག་བྱ་བ་དང༌། ཕྱི་ནང་གསང་གསུམ་གྱི་མཆོད་པ་ཀུན་ཏུ་བཟང་པོའི་མཆོད་སྤྲིན་གྱིས་ཡང་དག་པར་མཆོད་ལ། སྡིག་པ་བཤགས་པ་སོགས་བདུན་རྣམ་དག་བྱ། མཎྜལ་འབུལ་ཞིང་གསོལ་བ་དྲག་ཏུ་གདབ་སྟེ། དུས་གསུམ་གྱི་སངས་རྒྱས་ཐམས་ཅད་ཀྱི་སྐུ་གསུང་ཐུགས་མཉམ་པ་ཉིད་ལྷུན་གྲུབ་རྒྱུན་ཆད་མེད་པའི་ཡོན་ཏན་གྱི་གཏེར་དུ་གྱུར་པས། བདག་དང་ཁམས་གསུམ་གྱི་སེམས་ཅན་ཐམས་ཅད་ཀྱི་ཐ་མལ་གྱི་ལུས་ངག་ཡིད་གསུམ་རྣམ་པར་དག་ནས། བླ་མ་རིན་པོ་ཆེའི་སྐུ་གསུང་ཐུགས་དང་དབྱེར་མེད་པར་མཛད་གསོལ། ། ཞེས་གསོལ་བ་དུབ་དུབ་ཀྱི་བར་དུ་བརྒྱ་རྩའམ་སྟོང་ལ་སོགས་པ་བཏབ་པས། བླ་མའི་སྐུ་གསུང་ཐུགས་དང་རང་གི་ལུས་ངག་ཡིད་གསུམ་དབྱེར་མེད་གྱུར་པ་དེ་ཉིད་ཀྱི་ངང་རྩེ་གཅིག་ཏུ་བཞག་གོ །དེ་ནི་བླ་མ་སྤྲུལ་པའི་སྐུར་བསྒྲུབ་པ་སྟེ། དཔལ་ཕག་མོ་གྲུ་པའི་ཞལ་ནས། རྩ་བའི་བླ་མ་བཅོམ་ལྡན་བསྟན་པའི་གཙོ། ། དུས་གསུམ་སངས་རྒྱས་བྱང་ཆུབ་སེམས་དཔའ་དང༌། ། རྒྱུད་པའི་ཚོགས་བཅས་མཁའ་ལ་སྤྱན་དྲངས་ཏེ། ། སེང་གེ་པདྨ་ཉི་མ་ཟླ་བའི་གདན། ། སངས་རྒྱས་རོལ་པའི་རྣམ་པར་བཞུགས་པར་བསྒོམ། ། བདུན་པོ་དག་བྱ་ལུས་དང་ལོངས་སྤྱོད་དབུལ། ། བདག་ཉིད་རྣམ་དག་དབྱེར་མེད་གསོལ་བ་གདབ། ། རྣམ་གསུམ་སྐྱབས་འགྲོ་གུས་པས་གསང་སྔགས་བརྗོད། ། འཛིན་པ་གསུམ་བྲལ་དབྱེར་མེད་གྱུར་པར་བསམ། ། ཞེས་གསུངས་པའི་དོན་ནོ། ། གཉིས་པ་ལོངས་སྤྱོད་རྫོགས་པའི་སྐུར་བསྒྲུབ་པ་ནི། མདུན་གྱི་ནམ་མཁའི་ལྷ་ཚོགས་གོང་དུ་དམིགས་པ་དེ་དག་རང་ཉིད་ལ་ཐིམ་པས། སྙིང་པདྨ་འདབ་བརྒྱད་ཀྱི་ལྟེ་བར་ཉི་མ་དང་ཟླ་བའི་གདན་ལ། བླ་མ་ཉིད་སངས་རྒྱས་རྣམ་པར་སྣང་མཛད་ཀྱི་ངོ་བོའམ། དྲུག་པ་རྡོ་རྗེ་འཆང་ལྟ་བུ་སྐུ་ལ་ཞིང་ཁམས་ཐམས་ཅད་སྣང་ཞིང༌། ཞིང་ཁམས་ཐམས་ཅད་དུ་སྐུས་ཁྱབ་པ། འོད་དང་ཟེར་གྱི་ཚོགས་དང་ལྡན་པར་བསྒོམ། དེའི་གནས་གསུམ་དུ་ཡི་གེ་གསུམ་བཀོད་ལ་བླ་མ་བརྒྱུད་པའི་གསང་མཚན་བཟླས་པས། སྐུ་གསུང་ཐུགས་ལས་འོད་ཟེར་འཕྲོས་ཁམས་གསུམ་གྱི་སེམས་ཅན་གྱི་སྒྲིབ་པ་སྦྱངས། བླ་མའི་སྐུ་གསུང་ཐུགས་སུ་གྱུར་པས། འོད་ཟེར་དེ་དག་ཚུར་འདུས། བླ་མའི་གནས་གསུམ་དུ་ཐིམ། ཁམས་གསུམ་དེ་དག་འོད་ཀྱིས་གང་པར་དམིགས་ལ་བཟླས་པ་བྱའོ། །དེ་ནི་ལོངས་སྤྱོད་རྫོགས་པའི་སྐུར་བསྒྲུབ་པའོ། །དེ་ཡང་དཔལ་ཕ་མོ་གྲུ་པའི་ཞལ་ནས། དེ་ནས་ནམ་མཁར་སྤྲུལ་པའི་ལྷ་ཚོགས་རྣམས། ། བདག་ཉིད་ལ་ཐིམ་ཡི་དམ་སྐུར་གྱུར་ཏེ། ། ཐུགས་ཀྱི་དཀྱིལ་འཁོར་ཆུ་སྐྱེས་རྣམ་རྒྱས་ཤིང་། ། ཉི་མ་ཟླ་བ་རོལ་བའི་གདན་སྟེང་དུ། ། ཡིད་བཞིན་ནོར་བུ་བླ་མ་དམ་པའི་སྐུ། ། རྣམ་པར་སྣང་མཛད་འོད་ཟེར་རབ་འབར་བ། ། དེ་ཉིད་གནས་གསུམ་ཉིད་དུ་པདྨའི་སྟེང༌། ། སྐུ་གསུང་ཐུགས་ཀྱི་དེ་ཉིད་གསུམ་བཀོད་ལ། ། འབྲུ་གསུམ་ལས་བྱུང་ཟག་མེད་འོད་ཟེར་གྱིས། ། ཁམས་གསུམ་རྣམ་དག་དབྱེར་མེད་སྐུ་ཐོབ་ནས། ། དེ་ཉིད་ལ་ཐིམ་འོད་ཀྱི་དཀྱིལ་འཁོར་དུ། ། མི་འབྲལ་ཏིང་འཛིན་བསྐུལ་བྱེད་གསང་སྔགས་བརྗོད། ། ཞེས་གསུངས་པའི་དོན་ནོ། ། གསུམ་པ་ཆོས་ཀྱི་སྐུར་སྒྲུབ་པ་ནི། དེ་ཉིད་ཀྱི་ངང་ལས་སེམས་མ་བཅོས་པར་བཞག་སྟེ། ཕྱིའི་སྣང་བ་ལ་བཟོ་མི་བྱ། ནང་སེམས་ཉིད་ལ་ལས་མི་བཅོལ། སྒོམ་དུ་འཛིན་པའི་བློའི་དྲི་མ་བཤིག་ལ། ལུས་ངག་ཡིད་གསུམ་རང་ལུགས་སུ་གློད་ལ་བཞག །གདོད་ནས་གནས་པའི་སེམས་ཉིད་རྣམ་པར་དག་པ་འོད་གསལ་བ་ཕྱག་རྒྱ་ཆེན་པོ་དང༌། དེ་ཉིད་འགག་མེད་དུ་ཤར་བའི་བླ་མར་འཛིན་པའི་རྣམ་པར་རིག་པ་གཉིས་ཆུ་ལ་ཆུ་བཞག་པ་དང་མར་ལ་མར་བཞག་པ་བཞིན་དབྱེར་མེད་དུ་གྲོལ་བས། ལུས་ངག་ཡིད་གསུམ་གང་དུའངང་མི་གནས་ཏེ། སྐྱེ་བ་མེད་པ་ཆོས་སྐུ། འགག་པ་མེད་པ་ལ་ལོངས་སྐུ། གནས་པ་མེད་པ་སྤྲུལ་སྐུ། དབྱེར་མེད་པ་ངོ་བོ་ཉིད་ཀྱི་སྐུ་བླ་མ་དམ་པ་དེ་རང་གི་སེམས་ཀྱི་དེ་ཉིད་ཡིན་པས། དེ་ཉིད་མ་ཡེངས་པར་སྒྲུབ་པ་ནི་ཆོས་སྐུར་བསྒྲུབ་པ་སྟེ། དཔལ་ཕག་མོ་གྲུ་པའི་ཞལ་ནས། དོན་དམ་བློ་འདས་སེམས་ཉིད་ཆོས་ཀྱི་སྐུ། ། གདོད་ནས་རྣམ་དག་མ་སྐྱེས་འདུས་མ་བྱས། ། རང་བཞིན་འོད་གསལ་ལྷུན་གྲུབ་དེ་བཞིན་ཉིད། ། གློར་བུར་རྟོག་པའི་དྲི་མས་མ་བསླད་ཅིང༌། ། མི་རྟོག་ཟག་མེད་བདེ་ཆེན་ཞི་བའི་ངང༌། ། བསྒོམ་བྱ་སྒོམ་བྱེད་རྩོལ་བ་རྣམས་སྤངས་ཏེ། ། འབྲལ་མེད་ཆེན་པོར་མ་བཅོས་ལྷུག་པར་བཞག ། བརྗོད་བྲལ་བློ་འདས་ཕྱག་རྒྱ་ཆེན་པོ་ནི། ། སྐུ་གསུམ་དབྱེར་མེད་བླ་མ་དམ་པ་སྟེ། ། དེས་ནི་བདུད་དང་ཉོན་མོངས་ལས་རྒྱལ་ཞིང༌། ། དཔག་མེད་ཡོན་ཏན་རྒྱ་མཚོའི་བདག་ཉིད་ཀྱིས། ། སྒྲུབ་མཆོག་བླ་ན་མེད་པར་རྒྱལ་བས་བཤད། ། ཅེས་གསུངས་པའི་དོན་ཡིན་ཅིང་། ཨ་ཏི་བཀོད་པའི་རྒྱུད་ལས། དྲིན་དང་ལྡན་པའི་བླ་མ་དེ། ། སྤྱི་བོའི་གཙུག་གམ་ལག་པའི་མཐིལ། ། སྙིང་གི་དབུས་སུ་སུས་བསྒོམ་པ། ། སངས་རྒྱས་སྟོང་གི་ཡོན་ཏན་ཀྱང༌། ། གང་ཟག་དེ་ཡིས་འཛིན་པར་འགྱུར། ། ཅེས་གསུངས་པས། ཡོན་ཏན་ཐམས་ཅད་སྐྱེད་པར་བྱེད་པ་རྩ་བའི་བླ་མ་ཡིན་པས། དེའི་སྐུ་གསུང་ཐུགས་ཀྱི་མཛད་པ་རྣམས་ཡོན་ཏན་དུ་ལྟ་ཞིང༌། ཅི་གསུང་ཚད་མར་འཛིན་པ་གཅིག་གལ་ཆེ་སྟེ། དགྱེས་རྡོར་ལས། གཞན་གྱིས་བརྗོད་མིན་ལྷན་ཅིག་སྐྱེས། ། གང་དུ་ཡང་ནི་མི་རྙེད་དེ། ། བླ་མི་དུས་ཐབས་བརྟེན་པ་དང༌། ། བདག་གིས་བསོད་ནམས་ལས་ཤེས་བྱ། ། ཞེས་དང༌། དམ་ཚིག་བཀོད་པའི་རྒྱུད་ལས། རྫོགས་པའི་རིམ་པ་བསྐལ་པའི་བར་དུ་ནི། ། འདུ་འཛི་གཡེང་བ་སྤངས་ནས་བསྒོམས་པ་བས། ། བླ་མ་རང་སེམས་དཀྱིལ་དུ་དར་གཅིག་ཙམ། ། འབུམ་ཕྲག་ཉི་ཤུའི་ཆར་ཡང་མི་ཕོད་དོ། ། ཞེས་དང༌། དཔལ་གསང་བ་འདུས་པ་ལ། སློབ་དཔོན་སྙིང་གར་རབ་བསྒོམས་ན། ། བྱང་ཆུབ་འདི་ཡིས་ཐོབ་འགྱུར་ཞིང་། ། ངོ་མཚར་ཆེན་པོའི་དངོས་གྲུབ་ཡིན། ། སྐུ་གསུང་ཐུགས་ནི་འགྲུབ་པར་འགྱུར། ། ཅེས་གསུངས་སོ། །
